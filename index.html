Static IP / DHCP switcher (menu-driven script)

Self-elevates to administrator, then lets the user set a static IP, subnet mask and gateway, or switch the Ethernet adapter back to DHCP with release and renew. Used to change network settings quickly during support work.

$isAdmin = ([Security.Principal.WindowsPrincipal] `
    [Security.Principal.WindowsIdentity]::GetCurrent()
).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)

if (-not $isAdmin) {
    Start-Process powershell.exe -Verb RunAs `
        -ArgumentList "-NoProfile -ExecutionPolicy Bypass -File `"$PSCommandPath`""
    exit
}

$InterfaceName = "Ethernet"
Clear-Host
Write-Host "Ethernet IP Configuration" -ForegroundColor Cyan
Write-Host "1. Set Static IP"
Write-Host "2. Set Automatic DHCP"
$Choice = Read-Host "Select option"

if ($Choice -eq "1") {
    $IP      = Read-Host "Enter IP Address"
    $Mask    = Read-Host "Enter Subnet Mask"
    $Gateway = Read-Host "Enter Default Gateway"

    netsh interface ip set address name="$InterfaceName" static $IP $Mask $Gateway
    Write-Host "Static IP set successfully. IP: $IP" -ForegroundColor Green
}
elseif ($Choice -eq "2") {
    # Enable DHCP for address and DNS
    netsh interface ip set address name="$InterfaceName" source=dhcp
    netsh interface ip set dns name="$InterfaceName" source=dhcp

    # Release old IP, request a new one
    ipconfig /release "$InterfaceName"
    Start-Sleep -Seconds 2
    ipconfig /renew "$InterfaceName"
    Write-Host "Ethernet is now using Automatic DHCP." -ForegroundColor Green
}
else {
    Write-Host "Invalid option." -ForegroundColor Red
}

Write-Host "Current Ethernet Configuration:" -ForegroundColor Cyan
ipconfig
Read-Host "Press Enter to exit"
